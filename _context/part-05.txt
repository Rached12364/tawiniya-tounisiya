    <Routes>

          }
        />

          }

        <Route
          path="actualites"

              <ActualitesPage />
            </ProtectedRoute>
          }
        />
        {/* Notifications (tous roles) */}
        <Route
          path="notifications"
          element={
            <ProtectedRoute>
              <NotificationsPage />
            </ProtectedRoute>
          }
        />
        {/* Profil personnel (technicien connecte) */}
        <Route
          path="profil"

              <ProfilPage />
            </ProtectedRoute>
          }
        />
        {/* Espaces connectes (TASK-F007/F008) */}
        <Route
          path="espace/technicien"
